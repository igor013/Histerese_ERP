import api from "./api";

// ====================================================
// 📦 Service: GRUPOS
// ====================================================

export async function listarGrupos({ page = 1, limit = 100, q = "" } = {}) {
    const params = new URLSearchParams();
    params.set("page", page);
    params.set("limit", limit);
    if (q) params.set("q", q);

    const { data } = await api.get(`/grupos?${params.toString()}`);

    // 🔥 Retorna apenas o array de grupos
    return Array.isArray(data.items) ? data.items : [];
}
